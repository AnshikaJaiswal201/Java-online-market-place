1. Executive Summary & Architecture

The ShopVerse Online Market Place System is an enterprise-grade multi-vendor e-commerce platform designed to bridge Indian local merchants (sellers) and buyers (customers) on a single unified portal. Built around native Indian commercial categories—Electronics, Traditional Wear & Sarees, Spices & Herbs, and Handicrafts—the platform operates as a modern 3-Tier Enterprise Application:

  • Presentation Tier (Client SPA): Angular 14 with reactive forms, client-side route guards, and JWT HTTP interception.
  • Application Tier (REST API): Java 17 and Spring Boot 3.2.5 with stateless Spring Security 6, BCrypt encryption, and ACID-compliant @Transactional business logic.
  • Data Tier (Relational Storage): Local MySQL 8.0 server under the shopverse_db database in 3NF across 9 interconnected tables.

6. Backend Source Code (Spring Boot 3 + Java 17)

6.1 Security Configuration (SecurityConfig.java)

@Configuration
@EnableWebSecurity
@EnableMethodSecurity
public class SecurityConfig {
    @Bean
    public PasswordEncoder passwordEncoder() { return new BCryptPasswordEncoder(); }

    @Bean
    public SecurityFilterChain securityFilterChain(HttpSecurity http) throws Exception {
        http.csrf(csrf -> csrf.disable()).cors(Customizer.withDefaults())
            .sessionManagement(s -> s.sessionCreationPolicy(SessionCreationPolicy.STATELESS))
            .authorizeHttpRequests(auth -> auth
                .requestMatchers("/api/auth/**", "/api/products/**", "/api/categories/**").permitAll()
                .requestMatchers("/api/seller/**").hasAuthority("ROLE_SELLER")
                .requestMatchers("/api/cart/**").hasAuthority("ROLE_CUSTOMER")
                .requestMatchers("/api/orders/**").authenticated()
                .anyRequest().authenticated()
            );
        http.addFilterBefore(jwtAuthenticationFilter, UsernamePasswordAuthenticationFilter.class);
        return http.build();
    }
}

6.2 User Registration & BCrypt Service (AuthServiceImpl.java)

@Service
public class AuthServiceImpl implements AuthService {
    @Transactional
    public AuthResponse register(RegisterRequest request) {
        String email = request.getEmail().trim().toLowerCase();
        if (userRepository.existsByEmail(email)) {
            throw new BadRequestException("Email is already registered: " + email);
        }
        User user = new User();
        user.setEmail(email);
        user.setPassword(passwordEncoder.encode(request.getPassword()));
        user.setFullName(request.getFullName().trim());
        user.setIsActive(1);
        Role userRole = roleRepository.findByRoleName("ROLE_" + request.getRole().toUpperCase())
                .orElseGet(() -> roleRepository.save(new Role("ROLE_" + request.getRole().toUpperCase())));
        user.getRoles().add(userRole);
        User savedUser = userRepository.save(user);

        Authentication auth = authenticationManager.authenticate(
            new UsernamePasswordAuthenticationToken(email, request.getPassword()));
        String jwt = tokenProvider.generateToken(auth);
        return new AuthResponse(jwt, savedUser.getUserId(), savedUser.getEmail(), savedUser.getFullName(), List.of(userRole.getRoleName()));
    }
}

7.2 Authentication State Service (auth.service.ts)

@Injectable({ providedIn: 'root' })
export class AuthService {
  private apiUrl = `${environment.apiUrl}/auth`;
  private currentUserSubject = new BehaviorSubject(null);
  public currentUser$ = this.currentUserSubject.asObservable();

  login(credentials: any): Observable> {
    return this.http.post>(`${this.apiUrl}/login`, credentials);
  }

  register(userData: any): Observable> {
    this.logout();
    return this.http.post>(`${this.apiUrl}/register`, userData);
  }

  setSession(authData: AuthResponse): void {
    localStorage.setItem('auth_token', authData.token);
    localStorage.setItem('user_details', JSON.stringify(authData));
    this.currentUserSubject.next(authData);
  }
}
